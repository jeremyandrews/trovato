    {% if current_page > 1 %}
  • {{ t(s="Previous page") }}
    • {% endif %} {% if current_page > window + 1 %}
  • {{ 1 }}
    • {% if current_page > window + 2 %}
  • {{ "…" }}
    • {% endif %} {% endif %} {% for page in range(start=1, end=total_pages + 1) %} {% if page >= current_page - window and page <= current_page + window %}
  • {% if page == current_page %} {{ page }} {% else %} {{ page }} {% endif %}
    • {% endif %} {% endfor %} {% if current_page < total_pages - window %} {% if current_page < total_pages - window - 1 %}
  • {{ "…" }}
    • {% endif %}
  • {{ total_pages }}
    • {% endif %} {% if current_page < total_pages %}
  • {{ t(s="Next page") }}
    • {% endif %}
{% endif %} {% endmacro %}
